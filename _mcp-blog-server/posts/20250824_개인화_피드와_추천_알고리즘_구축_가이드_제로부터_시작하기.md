## 📊 Phase 1: 데이터 수집 인프라 구축 (1주일)

### 핵심 원칙: "추천보다 데이터가 먼저다"

아무리 좋은 알고리즘도 데이터 없이는 무용지물입니다. **가장 먼저 해야 할 일은 사용자 행동 데이터를 수집하는 시스템 구축**입니다.

### 이벤트 트래킹 시스템 구현

```typescript
// backend/src/analytics/events.service.ts

@Injectable()
export class EventTrackingService {
  constructor(
    @InjectRepository(UserEvent)
    private eventRepository: Repository<UserEvent>,
  ) {}

  async trackEvent(userId: string, event: UserEvent) {
    // 모든 사용자 행동을 데이터베이스에 저장
    await this.eventRepository.save({
      userId,
      eventType: event.type,
      targetId: event.targetId,
      metadata: event.metadata,
      timestamp: new Date(),
      sessionId: event.sessionId,
      // 추가 컨텍스트 정보
      deviceType: event.deviceType,
      referrer: event.referrer,
      location: event.location,
    });
    
    // 실시간 처리가 필요한 경우 Redis에도 저장
    await this.redis.zadd(
      `user:${userId}:recent_events`,
      Date.now(),
      JSON.stringify(event)
    );
  }
}
```

```typescript
// frontend/src/hooks/useEventTracking.ts

export const useEventTracking = () => {
  const { user } = useAuth();
  
  const trackEvent = useCallback(async (
    eventType: EventType,
    targetId: string,
    metadata?: any
  ) => {
    if (!user) return;
    
    const event = {
      type: eventType,
      targetId,
      metadata: {
        ...metadata,
        timestamp: Date.now(),
        url: window.location.href,
        viewport: {
          width: window.innerWidth,
          height: window.innerHeight,
        },
      },
      sessionId: getSessionId(),
      deviceType: getDeviceType(),
    };
    
    // 배치 처리를 위해 로컬 큐에 저장
    eventQueue.push(event);
    
    // 5초마다 또는 10개 이상 쌓이면 서버로 전송
    if (eventQueue.length >= 10) {
      await flushEvents();
    }
  }, [user]);
  
  // 스크롤 깊이 추적
  const trackScrollDepth = useCallback((postId: string) => {
    let maxDepth = 0;
    
    const handleScroll = throttle(() => {
      const scrollPercentage = 
        (window.scrollY + window.innerHeight) / 
        document.documentElement.scrollHeight * 100;
      
      if (scrollPercentage > maxDepth) {
        maxDepth = scrollPercentage;
        
        // 25%, 50%, 75%, 100% 지점에서 이벤트 발생
        if ([25, 50, 75, 100].includes(Math.floor(maxDepth))) {
          trackEvent(EventType.POST_SCROLL_DEPTH, postId, {
            depth: Math.floor(maxDepth)
          });
        }
      }
    }, 1000);
    
    window.addEventListener('scroll', handleScroll);
    return () => window.removeEventListener('scroll', handleScroll);
  }, [trackEvent]);
  
  return { trackEvent, trackScrollDepth };
};
```

## 🤖 Phase 3: 기본 추천 알고리즘 구현 (1주일)

### Step 1: 협업 필터링 (Collaborative Filtering)

**"당신과 비슷한 사용자들이 좋아한 콘텐츠"**

```typescript
// backend/src/recommendation/collaborative-filtering.service.ts

@Injectable()
export class CollaborativeFilteringService {
  
  // 사용자 간 유사도 계산 (코사인 유사도)
  async calculateUserSimilarity(userId1: string, userId2: string): Promise<number> {
    // 두 사용자가 공통으로 평가한 아이템 찾기
    const commonItems = await this.db.query(`
      SELECT 
        i1.post_id,
        i1.interaction_value as rating1,
        i2.interaction_value as rating2
      FROM user_interactions i1
      JOIN user_interactions i2 ON i1.post_id = i2.post_id
      WHERE i1.user_id = $1 
        AND i2.user_id = $2
        AND i1.interaction_type = 'rating'
        AND i2.interaction_type = 'rating'
    `, [userId1, userId2]);
    
    if (commonItems.length < 3) return 0; // 최소 3개 이상 공통 평가 필요
    
    // 코사인 유사도 계산
    let dotProduct = 0;
    let norm1 = 0;
    let norm2 = 0;
    
    for (const item of commonItems) {
      dotProduct += item.rating1 * item.rating2;
      norm1 += item.rating1 ** 2;
      norm2 += item.rating2 ** 2;
    }
    
    return dotProduct / (Math.sqrt(norm1) * Math.sqrt(norm2));
  }
  
  // 유사한 사용자 찾기
  async findSimilarUsers(userId: string, limit: number = 50): Promise<SimilarUser[]> {
    // 후보 사용자 찾기 (같은 포스트를 본 사용자들)
    const candidates = await this.db.query(`
      SELECT DISTINCT u2.user_id
      FROM user_interactions u1
      JOIN user_interactions u2 ON u1.post_id = u2.post_id
      WHERE u1.user_id = $1 
        AND u2.user_id != $1
      LIMIT 500
    `, [userId]);
    
    // 각 후보와의 유사도 계산
    const similarities = await Promise.all(
      candidates.map(async (candidate) => ({
        userId: candidate.user_id,
        similarity: await this.calculateUserSimilarity(userId, candidate.user_id)
      }))
    );
    
    // 유사도 높은 순으로 정렬
    return similarities
      .filter(s => s.similarity > 0.5) // 최소 유사도 임계값
      .sort((a, b) => b.similarity - a.similarity)
      .slice(0, limit);
  }
  
  // 협업 필터링 기반 추천
  async getRecommendations(userId: string, limit: number = 20): Promise<Post[]> {
    // 1. 유사한 사용자들 찾기
    const similarUsers = await this.findSimilarUsers(userId);
    
    if (similarUsers.length === 0) {
      // 콜드 스타트: 인기 콘텐츠 반환
      return this.getPopularPosts(limit);
    }
    
    // 2. 유사한 사용자들이 좋아한 포스트 중 내가 안 본 것들
    const recommendations = await this.db.query(`
      WITH user_reads AS (
        SELECT DISTINCT post_id 
        FROM user_interactions 
        WHERE user_id = $1
      ),
      similar_user_likes AS (
        SELECT 
          ui.post_id,
          SUM(ui.interaction_value * $2[idx]) as weighted_score
        FROM user_interactions ui
        JOIN unnest($3::uuid[]) WITH ORDINALITY as u(user_id, idx) 
          ON ui.user_id = u.user_id
        WHERE ui.interaction_type IN ('like', 'read_complete')
          AND ui.post_id NOT IN (SELECT post_id FROM user_reads)
        GROUP BY ui.post_id
      )
      SELECT p.*, sul.weighted_score
      FROM similar_user_likes sul
      JOIN posts p ON p.id = sul.post_id
      WHERE p.status = 'published'
      ORDER BY sul.weighted_score DESC
      LIMIT $4
    `, [
      userId,
      similarUsers.map(u => u.similarity),
      similarUsers.map(u => u.userId),
      limit
    ]);
    
    return recommendations;
  }
}
```

### Step 2: 콘텐츠 기반 필터링 (Content-Based)

**"당신이 좋아한 콘텐츠와 비슷한 콘텐츠"**

```typescript
// backend/src/recommendation/content-based.service.ts

@Injectable()
export class ContentBasedFilteringService {
  
  // TF-IDF 벡터 생성
  async createTfIdfVector(content: string): Promise<number[]> {
    // 형태소 분석 (한국어의 경우)
    const tokens = await this.tokenizer.tokenize(content);
    
    // 불용어 제거
    const filteredTokens = tokens.filter(t => !this.stopWords.includes(t));
    
    // TF 계산
    const tf = {};
    filteredTokens.forEach(token => {
      tf[token] = (tf[token] || 0) + 1 / filteredTokens.length;
    });
    
    // IDF 계산 (사전 계산된 값 사용)
    const tfidf = {};
    for (const [token, tfValue] of Object.entries(tf)) {
      const idf = await this.getIdf(token);
      tfidf[token] = tfValue * idf;
    }
    
    // 벡터로 변환
    return this.vectorize(tfidf);
  }
  
  // 임베딩 기반 유사도 (OpenAI API 사용)
  async createEmbedding(content: string): Promise<number[]> {
    const response = await this.openai.createEmbedding({
      model: "text-embedding-ada-002",
      input: content.slice(0, 8000), // 토큰 제한
    });
    
    return response.data[0].embedding;
  }
  
  // 코사인 유사도 계산
  calculateCosineSimilarity(vec1: number[], vec2: number[]): number {
    let dotProduct = 0;
    let norm1 = 0;
    let norm2 = 0;
    
    for (let i = 0; i < vec1.length; i++) {
      dotProduct += vec1[i] * vec2[i];
      norm1 += vec1[i] ** 2;
      norm2 += vec2[i] ** 2;
    }
    
    return dotProduct / (Math.sqrt(norm1) * Math.sqrt(norm2));
  }
  
  // 콘텐츠 기반 추천
  async getContentBasedRecommendations(
    userId: string, 
    limit: number = 20
  ): Promise<Post[]> {
    // 1. 사용자가 최근 좋아한/읽은 포스트
    const userHistory = await this.getUserPreferredPosts(userId, 10);
    
    if (userHistory.length === 0) {
      return []; // 콜드 스타트
    }
    
    // 2. 사용자 프로필 벡터 생성 (좋아한 포스트들의 평균)
    const userVector = await this.createUserProfileVector(userHistory);
    
    // 3. 벡터 유사도 기반 검색 (pgvector 사용)
    const recommendations = await this.db.query(`
      WITH user_reads AS (
        SELECT DISTINCT post_id 
        FROM user_interactions 
        WHERE user_id = $1
      )
      SELECT 
        p.*,
        1 - (pf.content_embedding <=> $2::vector) as similarity
      FROM posts p
      JOIN post_features pf ON p.id = pf.post_id
      WHERE p.id NOT IN (SELECT post_id FROM user_reads)
        AND p.status = 'published'
      ORDER BY similarity DESC
      LIMIT $3
    `, [userId, userVector, limit]);
    
    return recommendations;
  }
  
  // 태그/카테고리 기반 추천 (간단한 버전)
  async getTagBasedRecommendations(
    userId: string,
    limit: number = 20
  ): Promise<Post[]> {
    // 사용자가 선호하는 태그 가져오기
    const preferredTags = await this.db.query(`
      SELECT 
        unnest(p.tags) as tag,
        COUNT(*) * AVG(ui.interaction_value) as score
      FROM user_interactions ui
      JOIN posts p ON ui.post_id = p.id
      WHERE ui.user_id = $1
        AND ui.interaction_type IN ('like', 'read_complete')
      GROUP BY tag
      ORDER BY score DESC
      LIMIT 10
    `, [userId]);
    
    // 선호 태그가 포함된 포스트 추천
    return this.db.query(`
      WITH user_reads AS (
        SELECT DISTINCT post_id 
        FROM user_interactions 
        WHERE user_id = $1
      )
      SELECT DISTINCT p.*
      FROM posts p
      WHERE p.tags && $2::text[]  -- 태그 배열 교집합
        AND p.id NOT IN (SELECT post_id FROM user_reads)
        AND p.status = 'published'
      ORDER BY 
        cardinality(p.tags & $2::text[]) DESC, -- 매칭되는 태그 수
        p.created_at DESC
      LIMIT $3
    `, [userId, preferredTags.map(t => t.tag), limit]);
  }
}
```

### 여러 추천 방식을 조합한 최종 시스템

```typescript
// backend/src/recommendation/hybrid-recommendation.service.ts

@Injectable()
export class HybridRecommendationService {
  constructor(
    private collaborative: CollaborativeFilteringService,
    private contentBased: ContentBasedFilteringService,
    private trending: TrendingService,
    private diversity: DiversityService,
  ) {}
  
  // 메인 추천 엔진
  async generatePersonalizedFeed(
    userId: string,
    page: number = 1,
    limit: number = 20,
    context?: FeedContext
  ): Promise<PersonalizedFeed> {
    // 사용자 프로필 로드
    const userProfile = await this.getUserProfile(userId);
    
    // 1. 다양한 소스에서 후보 추천 가져오기
    const [
      collaborativeRecs,
      contentRecs,
      trendingPosts,
      followingPosts,
      newPosts
    ] = await Promise.all([
      // 협업 필터링: 유사 사용자 기반
      this.collaborative.getRecommendations(userId, 50),
      
      // 콘텐츠 기반: 선호 콘텐츠 유사성
      this.contentBased.getRecommendations(userId, 50),
      
      // 트렌딩: 최근 인기 콘텐츠
      this.trending.getTrendingPosts(30, context?.timeWindow),
      
      // 팔로잉: 팔로우한 작가의 새 글
      this.getFollowingAuthorPosts(userId, 20),
      
      // 신규: 최근 발행된 고품질 콘텐츠
      this.getNewQualityPosts(20)
    ]);
    
    // 2. 점수 계산 및 가중치 적용
    const scoredPosts = await this.calculateHybridScores({
      collaborative: { 
        posts: collaborativeRecs, 
        weight: this.getWeight('collaborative', userProfile) 
      },
      content: { 
        posts: contentRecs, 
        weight: this.getWeight('content', userProfile) 
      },
      trending: { 
        posts: trendingPosts, 
        weight: this.getWeight('trending', userProfile) 
      },
      following: { 
        posts: followingPosts, 
        weight: this.getWeight('following', userProfile) 
      },
      new: { 
        posts: newPosts, 
        weight: this.getWeight('new', userProfile) 
      }
    }, userProfile);
    
    // 3. 다양성 보장 알고리즘
    const diversifiedFeed = await this.diversity.ensureDiversity(
      scoredPosts,
      {
        maxSameAuthor: 2,      // 같은 작가 최대 2개
        maxSameTag: 3,         // 같은 태그 연속 3개까지
        maxSameCategory: 4,    // 같은 카테고리 최대 4개
        minReadingTimeVariety: true, // 읽기 시간 다양성
        includeSerendipity: 0.1, // 10% 뜻밖의 발견
      }
    );
    
    // 4. 재정렬 (시간대, 컨텍스트 고려)
    const rerankedFeed = await this.rerank(diversifiedFeed, {
      timeOfDay: new Date().getHours(),
      dayOfWeek: new Date().getDay(),
      userContext: context,
      userProfile
    });
    
    // 5. 페이지네이션 및 캐싱
    const paginatedFeed = this.paginate(rerankedFeed, page, limit);
    
    // 캐시 저장 (다음 페이지 미리 준비)
    await this.cacheNextPage(userId, rerankedFeed, page + 1, limit);
    
    // 6. 메타데이터 추가
    return {
      posts: paginatedFeed,
      metadata: {
        page,
        hasMore: rerankedFeed.length > page * limit,
        totalCount: rerankedFeed.length,
        generatedAt: new Date(),
        algorithms: ['collaborative', 'content', 'trending'],
        personalizationScore: this.calculatePersonalizationScore(paginatedFeed, userProfile)
      }
    };
  }
  
  // 동적 가중치 계산
  private getWeight(algorithm: string, userProfile: UserProfile): number {
    // 사용자 활동 패턴에 따라 가중치 조정
    const baseWeights = {
      collaborative: 0.3,
      content: 0.25,
      trending: 0.2,
      following: 0.15,
      new: 0.1
    };
    
    // 신규 사용자는 트렌딩과 새 콘텐츠 비중 높임
    if (userProfile.totalReads < 10) {
      return algorithm === 'trending' ? 0.4 : 
             algorithm === 'new' ? 0.3 : 
             baseWeights[algorithm] * 0.5;
    }
    
    // 활발한 사용자는 협업 필터링 비중 높임
    if (userProfile.engagementScore > 0.7) {
      return algorithm === 'collaborative' ? 0.4 : 
             baseWeights[algorithm];
    }
    
    return baseWeights[algorithm];
  }
  
  // 하이브리드 점수 계산
  private async calculateHybridScores(
    sources: RecommendationSources,
    userProfile: UserProfile
  ): Promise<ScoredPost[]> {
    const postScores = new Map<string, number>();
    const postDetails = new Map<string, Post>();
    
    // 각 소스별로 점수 누적
    for (const [source, data] of Object.entries(sources)) {
      data.posts.forEach((post, index) => {
        const postId = post.id;
        postDetails.set(postId, post);
        
        // 순위 기반 점수 (상위 랭크일수록 높은 점수)
        const rankScore = 1 / (index + 1);
        
        // 가중치 적용
        const weightedScore = rankScore * data.weight;
        
        // 누적 점수
        const currentScore = postScores.get(postId) || 0;
        postScores.set(postId, currentScore + weightedScore);
      });
    }
    
    // 추가 신호 반영
    for (const [postId, score] of postScores.entries()) {
      const post = postDetails.get(postId);
      
      // 최신성 보너스
      const ageInDays = (Date.now() - post.createdAt.getTime()) / (1000 * 60 * 60 * 24);
      const freshnessBonus = Math.exp(-ageInDays / 7) * 0.1; // 일주일 반감기
      
      // 품질 점수 보너스
      const qualityBonus = (post.qualityScore || 0) * 0.1;
      
      // 사용자 선호도 보너스
      const preferenceBonus = await this.calculatePreferenceScore(post, userProfile);
      
      postScores.set(postId, score + freshnessBonus + qualityBonus + preferenceBonus);
    }
    
    // 정렬 및 반환
    return Array.from(postScores.entries())
      .map(([postId, score]) => ({
        ...postDetails.get(postId),
        score
      }))
      .sort((a, b) => b.score - a.score);
  }
}
```

```typescript
// 가장 간단한 추천: 인기도 + 최신 믹스
class SimpleRecommendationService {
  async getBasicFeed(userId?: string, page: number = 1) {
    const limit = 20;
    const offset = (page - 1) * limit;
    
    // 1. 인기 포스트 (좋아요 + 조회수 기반)
    const popularPosts = await this.db.query(`
      SELECT p.*, 
        (p.like_count * 2 + p.view_count * 0.1 + p.comment_count * 3) as popularity_score
      FROM posts p
      WHERE p.created_at > NOW() - INTERVAL '7 days'
        AND p.status = 'published'
      ORDER BY popularity_score DESC
      LIMIT $1 OFFSET $2
    `, [limit / 2, offset / 2]);
    
    // 2. 최신 포스트
    const recentPosts = await this.db.query(`
      SELECT p.*
      FROM posts p
      WHERE p.status = 'published'
      ORDER BY p.created_at DESC
      LIMIT $1 OFFSET $2
    `, [limit / 2, offset / 2]);
    
    // 3. 사용자별 필터링 (로그인한 경우)
    if (userId) {
      const readPosts = await this.getReadPostIds(userId);
      const combined = [...popularPosts, ...recentPosts]
        .filter(p => !readPosts.includes(p.id));
      
      // 4. 간단한 개인화: 선호 태그 부스트
      const preferredTags = await this.getUserPreferredTags(userId);
      return this.boostByTags(combined, preferredTags);
    }
    
    // 5. 랜덤 셔플 믹스
    return this.shuffleWithWeight(popularPosts, recentPosts);
  }
  
  // 태그 기반 부스트
  private boostByTags(posts: Post[], preferredTags: string[]): Post[] {
    return posts.map(post => {
      const tagMatch = post.tags.filter(t => preferredTags.includes(t)).length;
      return {
        ...post,
        score: (post.score || 0) + tagMatch * 0.2
      };
    }).sort((a, b) => b.score - a.score);
  }
  
  // 가중치 기반 셔플
  private shuffleWithWeight(popular: Post[], recent: Post[]): Post[] {
    const result = [];
    let popularIndex = 0;
    let recentIndex = 0;
    
    // 2:1 비율로 믹스 (인기 2개, 최신 1개)
    while (popularIndex < popular.length || recentIndex < recent.length) {
      if (popularIndex < popular.length) {
        result.push(popular[popularIndex++]);
      }
      if (popularIndex < popular.length) {
        result.push(popular[popularIndex++]);
      }
      if (recentIndex < recent.length) {
        result.push(recent[recentIndex++]);
      }
    }
    
    return result;
  }
}
```

### 핵심 KPI

```typescript
// 추천 시스템 성과 측정
class RecommendationMetrics {
  
  // 1. 참여도 지표
  async calculateEngagementRate(userId: string, period: string): Promise<number> {
    const result = await this.db.query(`
      SELECT 
        COUNT(DISTINCT post_id) as viewed_posts,
        COUNT(CASE WHEN interaction_type = 'like' THEN 1 END) as likes,
        COUNT(CASE WHEN interaction_type = 'comment' THEN 1 END) as comments,
        COUNT(CASE WHEN interaction_type = 'share' THEN 1 END) as shares
      FROM user_interactions
      WHERE user_id = $1 
        AND created_at > NOW() - INTERVAL $2
    `, [userId, period]);
    
    const engagementActions = result.likes + result.comments + result.shares;
    return engagementActions / Math.max(result.viewed_posts, 1);
  }
  
  // 2. 클릭률 (CTR)
  async calculateCTR(recommendationBatch: string): Promise<number> {
    const result = await this.db.query(`
      SELECT 
        COUNT(*) as impressions,
        COUNT(CASE WHEN clicked = true THEN 1 END) as clicks
      FROM recommendation_logs
      WHERE batch_id = $1
    `, [recommendationBatch]);
    
    return result.clicks / Math.max(result.impressions, 1);
  }
  
  // 3. 체류 시간
  async calculateDwellTime(userId: string): Promise<number> {
    return this.db.query(`
      SELECT AVG(
        EXTRACT(EPOCH FROM (last_interaction - first_interaction))
      ) as avg_dwell_time
      FROM session_logs
      WHERE user_id = $1 
        AND last_interaction > NOW() - INTERVAL '30 days'
    `, [userId]);
  }
  
  // 4. 다양성 점수
  calculateDiversityScore(recommendations: Post[]): number {
    const uniqueTags = new Set();
    const uniqueAuthors = new Set();
    const uniqueCategories = new Set();
    
    recommendations.forEach(post => {
      post.tags.forEach(tag => uniqueTags.add(tag));
      uniqueAuthors.add(post.authorId);
      uniqueCategories.add(post.category);
    });
    
    // Simpson's Diversity Index
    const tagDiversity = uniqueTags.size / Math.max(recommendations.length, 1);
    const authorDiversity = uniqueAuthors.size / Math.max(recommendations.length, 1);
    const categoryDiversity = uniqueCategories.size / Math.max(recommendations.length, 1);
    
    return (tagDiversity + authorDiversity + categoryDiversity) / 3;
  }
  
  // 5. 신규 발견률
  async calculateDiscoveryRate(userId: string): Promise<number> {
    const result = await this.db.query(`
      WITH user_usual_tags AS (
        SELECT unnest(array_agg(DISTINCT tag)) as tag
        FROM (
          SELECT unnest(p.tags) as tag
          FROM user_interactions ui
          JOIN posts p ON ui.post_id = p.id
          WHERE ui.user_id = $1 
            AND ui.created_at < NOW() - INTERVAL '7 days'
        ) t
      ),
      recent_interactions AS (
        SELECT p.tags
        FROM user_interactions ui
        JOIN posts p ON ui.post_id = p.id
        WHERE ui.user_id = $1 
          AND ui.created_at > NOW() - INTERVAL '7 days'
          AND ui.interaction_type IN ('like', 'read_complete')
      )
      SELECT 
        COUNT(*) as total_interactions,
        COUNT(CASE 
          WHEN NOT (tags && (SELECT array_agg(tag) FROM user_usual_tags))
          THEN 1 
        END) as new_discoveries
      FROM recent_interactions
    `, [userId]);
    
    return result.new_discoveries / Math.max(result.total_interactions, 1);
  }
}
```
